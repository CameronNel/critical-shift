using System;using System.IO;using UnityEngine;using UnityEditor;using UnityEditor.SceneManagement;using UnityEngine.Rendering;
public static class CsMineGlassControlFinal11 {
 [Serializable] public sealed class Source {public float[] color,worldLinear;public float f0,roughness;}
 public static void Render(){
  string output="/tmp/cs-mine-glass-control-final11/";var source=JsonUtility.FromJson<Source>(File.ReadAllText(output+"source-provenance.json"));EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);RenderSettings.skybox=null;RenderSettings.ambientMode=AmbientMode.Flat;RenderSettings.ambientLight=Color.black;RenderSettings.ambientIntensity=0;RenderSettings.reflectionIntensity=1;RenderSettings.defaultReflectionMode=DefaultReflectionMode.Custom;
  var cube=new Cubemap(16,TextureFormat.RGBAFloat,false);var values=new Color[16*16];for(int i=0;i<values.Length;i++)values[i]=new Color(source.worldLinear[0],source.worldLinear[1],source.worldLinear[2],1);foreach(CubemapFace face in new[]{CubemapFace.PositiveX,CubemapFace.NegativeX,CubemapFace.PositiveY,CubemapFace.NegativeY,CubemapFace.PositiveZ,CubemapFace.NegativeZ})cube.SetPixels(values,face);cube.Apply(false,false);var property=typeof(RenderSettings).GetProperty("customReflectionTexture") ?? typeof(RenderSettings).GetProperty("customReflection");if(property==null)throw new Exception("No supported custom reflection setter");property.SetValue(null,cube);ReflectionProbe.UpdateCachedState();
  var pane=GameObject.CreatePrimitive(PrimitiveType.Quad);pane.transform.localScale=Vector3.one*10;var renderer=pane.GetComponent<MeshRenderer>();renderer.lightProbeUsage=LightProbeUsage.Off;renderer.reflectionProbeUsage=ReflectionProbeUsage.BlendProbes;var material=new Material(Shader.Find("CriticalShift/BlenderGlass"));renderer.sharedMaterial=material;material.SetColor("_Color",new Color(source.color[0],source.color[1],source.color[2],.12f).gamma);material.SetFloat("_Metallic",0);material.SetFloat("_MetallicFromMap",0);material.SetFloat("_SmoothnessFromMap",0);material.SetFloat("_Glossiness",1-source.roughness);
  var back=GameObject.CreatePrimitive(PrimitiveType.Quad);back.transform.position=Vector3.forward;back.transform.localScale=Vector3.one*20;var black=new Material(Shader.Find("Unlit/Color"));black.color=Color.black;back.GetComponent<MeshRenderer>().sharedMaterial=black;back.GetComponent<MeshRenderer>().lightProbeUsage=LightProbeUsage.Off;
  var camera=new GameObject("Glass diagnostic camera").AddComponent<Camera>();camera.orthographic=true;camera.orthographicSize=.4f;camera.allowHDR=true;camera.allowMSAA=false;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.black;camera.nearClipPlane=.01f;camera.renderingPath=RenderingPath.Forward;
  void Save(string name){var rt=new RenderTexture(128,128,24,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);camera.targetTexture=rt;camera.Render();var previous=RenderTexture.active;RenderTexture.active=rt;var pixels=new Texture2D(128,128,TextureFormat.RGBAFloat,false,true);pixels.ReadPixels(new Rect(0,0,128,128),0,0);pixels.Apply();foreach(var pixel in pixels.GetPixels())if(!float.IsFinite(pixel.r)||!float.IsFinite(pixel.g)||!float.IsFinite(pixel.b))throw new Exception("Nonfinite glass RGB");File.WriteAllBytes(output+name+".exr",pixels.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));RenderTexture.active=previous;camera.targetTexture=null;UnityEngine.Object.DestroyImmediate(pixels);UnityEngine.Object.DestroyImmediate(rt);}
  foreach(int angle in new[]{0,40,65}){float theta=angle*Mathf.Deg2Rad;camera.transform.position=new Vector3(Mathf.Sin(theta)*4,0,-Mathf.Cos(theta)*4);camera.transform.LookAt(Vector3.zero);material.shader=Shader.Find("CriticalShift/BlenderGlass");material.SetColor("_Color",new Color(source.color[0],source.color[1],source.color[2],.12f).gamma);Save("Native-fade-angle"+angle);material.shader=Shader.Find("CriticalShift/DiagnosticGlassPremul");material.SetColor("_Color",new Color(source.color[0],source.color[1],source.color[2],1).gamma);material.SetFloat("_SourceF0",source.f0);material.SetFloat("_Glossiness",1-source.roughness);Save("Native-premul-diagnostic-angle"+angle);}
  Debug.Log("CS_NATIVE_GLASS_REFLECTION_CONTROLS_OK; no production shader change; full BTDF/vehicle/reflection/Player unapproved");EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);UnityEngine.Object.DestroyImmediate(cube);
 }
}
