using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using UnityEditor.SceneManagement;
public static class CsMineShBasisFinal12 {
 [Serializable] public class Row {public Vector3 direction;public float cpuBase,gpuBase;public float[] cpuRawUnit,cpuOffsetBasis,gpuOffsetBasis;}
 [Serializable] public class Proof {public string status,shader,quantity,unityVersion,driver;public bool linearColour;public float receiverAlbedo,maxGpuCpuBasisError;public Row[] rows;}
 public static void Run(){
  const string output="/tmp/cs-mine-sh-basis-final12/";Directory.CreateDirectory(output);EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
  if(QualitySettings.activeColorSpace!=ColorSpace.Linear)throw new Exception("SH control needs Linear colourspace");RenderSettings.ambientMode=AmbientMode.Custom;RenderSettings.ambientProbe=new SphericalHarmonicsL2();RenderSettings.ambientIntensity=0;RenderSettings.reflectionIntensity=0;
  var renderer=new GameObject("Neutral custom-SH receiver").AddComponent<MeshRenderer>();var filter=renderer.gameObject.AddComponent<MeshFilter>();var mesh=new Mesh();filter.sharedMesh=mesh;renderer.lightProbeUsage=LightProbeUsage.CustomProvided;renderer.reflectionProbeUsage=ReflectionProbeUsage.Off;renderer.shadowCastingMode=ShadowCastingMode.Off;
  var shader=Shader.Find("CriticalShift/BlenderSurface");if(shader==null)throw new Exception("Missing actual source shader");var material=new Material(shader);material.SetColor("_Color",new Color(.18f,.18f,.18f).gamma);material.SetFloat("_Metallic",0);material.SetFloat("_DielectricSpecular",0);material.SetFloat("_Glossiness",0);renderer.sharedMaterial=material;
  var camera=new GameObject("SH basis camera").AddComponent<Camera>();camera.orthographic=true;camera.orthographicSize=.75f;camera.nearClipPlane=.05f;camera.farClipPlane=8;camera.allowHDR=true;camera.allowMSAA=false;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.black;camera.renderingPath=RenderingPath.Forward;
  var baseline=new SphericalHarmonicsL2();for(int colour=0;colour<3;colour++)baseline[colour,0]=3.5449077018f;
  var directions=new List<Vector3>();for(int x=-1;x<=1;x++)for(int y=-1;y<=1;y++)for(int z=-1;z<=1;z++)if(x!=0||y!=0||z!=0)directions.Add(new Vector3(x,y,z).normalized);
  int captureIndex=0;float Read(SphericalHarmonicsL2 sh){var block=new MaterialPropertyBlock();block.CopySHCoefficientArraysFrom(new[]{sh});renderer.SetPropertyBlock(block);var rt=new RenderTexture(16,16,24,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);camera.targetTexture=rt;camera.Render();var previous=RenderTexture.active;RenderTexture.active=rt;var pixels=new Texture2D(16,16,TextureFormat.RGBAFloat,false,true);pixels.ReadPixels(new Rect(0,0,16,16),0,0);pixels.Apply();foreach(var pixel in pixels.GetPixels())if(!float.IsFinite(pixel.r)||!float.IsFinite(pixel.g)||!float.IsFinite(pixel.b))throw new Exception("Nonfinite SH control HDR pixel");File.WriteAllBytes(output+"Basis-"+(captureIndex++).ToString("D3")+".exr",pixels.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));var colour=pixels.GetPixel(8,8);if(!float.IsFinite(colour.r)||!float.IsFinite(colour.g)||!float.IsFinite(colour.b)||Mathf.Abs(colour.r-colour.g)>.00001f||Mathf.Abs(colour.r-colour.b)>.00001f)throw new Exception("Nonfinite or asymmetric neutral SH response");camera.targetTexture=null;RenderTexture.active=previous;UnityEngine.Object.DestroyImmediate(pixels);UnityEngine.Object.DestroyImmediate(rt);return colour.r/.18f;}
  var rows=new List<Row>();float maxError=0;
  foreach(var n in directions){var tangent=Vector3.Cross(n,Mathf.Abs(n.y)<.95f?Vector3.up:Vector3.right).normalized;var b=Vector3.Cross(n,tangent).normalized;mesh.Clear();mesh.vertices=new[]{-tangent-b,tangent-b,tangent+b,-tangent+b};mesh.normals=Enumerable.Repeat(n,4).ToArray();mesh.uv=new[]{Vector2.zero,Vector2.right,Vector2.one,Vector2.up};mesh.triangles=new[]{0,1,2,0,2,3};mesh.RecalculateBounds();camera.transform.position=n*3;camera.transform.rotation=Quaternion.LookRotation(-n,b);
   var result=new Color[1];baseline.Evaluate(new[]{n},result);float cpuBase=result[0].r;if(!float.IsFinite(cpuBase)||!float.IsFinite(result[0].g)||!float.IsFinite(result[0].b))throw new Exception("Nonfinite CPU SH baseline");float gpuBase=Read(baseline);maxError=Mathf.Max(maxError,Mathf.Abs(cpuBase-gpuBase),Mathf.Abs(cpuBase-1),Mathf.Abs(gpuBase-1));var row=new Row{direction=n,cpuBase=cpuBase,gpuBase=gpuBase,cpuRawUnit=new float[9],cpuOffsetBasis=new float[9],gpuOffsetBasis=new float[9]};
   for(int index=0;index<9;index++){var unit=new SphericalHarmonicsL2();var offset=baseline;for(int colour=0;colour<3;colour++){unit[colour,index]=1;offset[colour,index]+=1;}unit.Evaluate(new[]{n},result);if(!float.IsFinite(result[0].r)||!float.IsFinite(result[0].g)||!float.IsFinite(result[0].b))throw new Exception("Nonfinite raw CPU SH RGB");row.cpuRawUnit[index]=result[0].r;offset.Evaluate(new[]{n},result);if(!float.IsFinite(result[0].r)||!float.IsFinite(result[0].g)||!float.IsFinite(result[0].b))throw new Exception("Nonfinite offset CPU SH RGB");row.cpuOffsetBasis[index]=result[0].r-cpuBase;row.gpuOffsetBasis[index]=Read(offset)-gpuBase;if(!float.IsFinite(row.cpuRawUnit[index])||!float.IsFinite(row.cpuOffsetBasis[index])||!float.IsFinite(row.gpuOffsetBasis[index]))throw new Exception("Nonfinite CPU/GPU SH coefficient");maxError=Mathf.Max(maxError,Mathf.Abs(row.gpuOffsetBasis[index]-row.cpuOffsetBasis[index]));}
   rows.Add(row);Debug.Log("CS_SH_BASIS_DIRECTION "+n+" baseCPU="+cpuBase+" baseGPU="+gpuBase);
  }
  File.WriteAllText(output+"basis.json",JsonUtility.ToJson(new Proof{status=maxError<.001f?"PassedBasisControlOnly":"RejectedBasisMismatch",shader=shader.name,unityVersion=Application.unityVersion,driver=SystemInfo.graphicsDeviceVersion,quantity="Signed irradiance-response basis recovered by positive offsets; no second cosine convolution",linearColour=true,receiverAlbedo=.18f,maxGpuCpuBasisError=maxError,rows=rows.ToArray()},true));
  if(!float.IsFinite(maxError)||maxError>=.001f)throw new Exception("Actual source-shader/CPU SH basis mismatch "+maxError);Debug.Log("CS_SH_GPU_CPU_BASIS_OK; no source/scene save, no actor/probe acceptance");EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
 }
}
