using System;using System.IO;using System.Linq;using System.Collections.Generic;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using UnityEditor.SceneManagement;
public static class CsMineShFittedGpuFinal12 {
 [Serializable] public class Coefficient {public Vector3 rgb;}
 [Serializable] public class Sample {public Vector3 direction,expectedClampedEOverPi;}
 [Serializable] public class Inputs {public Coefficient[] coefficients;public Sample[] samples;}
 [Serializable] public class Row {public int sample,albedoCase;public Vector3 normal,albedo,cpuSignedEOverPi,gpuOutgoingRGB,expectedOutgoingRGB;}
 [Serializable] public class Proof {public string status,unityVersion,driver,shader;public float maxAbsError;public Row[] rows;}
 public static void Run(){
  const string output="/tmp/cs-mine-sh-fitted-gpu-final12/";var input=JsonUtility.FromJson<Inputs>(File.ReadAllText(output+"inputs.json"));if(input.coefficients.Length!=9||input.samples.Length!=26)throw new Exception("Wrong fitted field input shape");
  EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);if(QualitySettings.activeColorSpace!=ColorSpace.Linear)throw new Exception("Expected Linear shader control");RenderSettings.ambientMode=AmbientMode.Custom;RenderSettings.ambientProbe=new SphericalHarmonicsL2();RenderSettings.ambientIntensity=0;RenderSettings.reflectionIntensity=0;
  var renderer=new GameObject("Actual fitted-SH neutral receiver").AddComponent<MeshRenderer>();renderer.lightProbeUsage=LightProbeUsage.CustomProvided;renderer.reflectionProbeUsage=ReflectionProbeUsage.Off;renderer.shadowCastingMode=ShadowCastingMode.Off;renderer.lightmapIndex=-1;var mesh=new Mesh();renderer.gameObject.AddComponent<MeshFilter>().sharedMesh=mesh;
  var shader=Shader.Find("CriticalShift/BlenderSurface");if(shader==null)throw new Exception("Missing actual source shader");var material=new Material(shader);material.SetFloat("_Metallic",0);material.SetFloat("_DielectricSpecular",0);material.SetFloat("_Glossiness",0);renderer.sharedMaterial=material;
  var sh=new SphericalHarmonicsL2();for(int j=0;j<9;j++){var c=input.coefficients[j].rgb;if(!Finite(c))throw new Exception("Nonfinite fitted coefficient");sh[0,j]=c.x;sh[1,j]=c.y;sh[2,j]=c.z;}var block=new MaterialPropertyBlock();block.CopySHCoefficientArraysFrom(new[]{sh});renderer.SetPropertyBlock(block);
  var camera=new GameObject("Fitted field verification camera").AddComponent<Camera>();camera.orthographic=true;camera.orthographicSize=.75f;camera.nearClipPlane=.05f;camera.farClipPlane=8;camera.allowHDR=true;camera.allowMSAA=false;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.black;camera.renderingPath=RenderingPath.Forward;
  var rows=new List<Row>();float maxError=0;var albedos=new[]{new Vector3(.18f,.18f,.18f),new Vector3(.5f,.5f,.5f),new Vector3(.1f,.25f,.6f)};
  for(int i=0;i<input.samples.Length;i++){
   var n=input.samples[i].direction;if(!Finite(n)||Mathf.Abs(n.magnitude-1)>.00001f)throw new Exception("Bad measured normal");var t=Vector3.Cross(n,Mathf.Abs(n.y)<.95f?Vector3.up:Vector3.right).normalized;var b=Vector3.Cross(n,t).normalized;mesh.Clear();mesh.vertices=new[]{-t-b,t-b,t+b,-t+b};mesh.normals=Enumerable.Repeat(n,4).ToArray();mesh.uv=new[]{Vector2.zero,Vector2.right,Vector2.one,Vector2.up};mesh.triangles=new[]{0,1,2,0,2,3};mesh.RecalculateBounds();camera.transform.position=n*3;camera.transform.rotation=Quaternion.LookRotation(-n,b);var cpu=new Color[1];sh.Evaluate(new[]{n},cpu);var cpuSigned=new Vector3(cpu[0].r,cpu[0].g,cpu[0].b);if(!Finite(cpuSigned))throw new Exception("Nonfinite CPU fitted field");
   for(int a=0;a<albedos.Length;a++){
    var albedo=albedos[a];material.SetColor("_Color",new Color(albedo.x,albedo.y,albedo.z).gamma);var rt=new RenderTexture(16,16,24,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);camera.targetTexture=rt;camera.Render();var previous=RenderTexture.active;RenderTexture.active=rt;var pixels=new Texture2D(16,16,TextureFormat.RGBAFloat,false,true);pixels.ReadPixels(new Rect(0,0,16,16),0,0);pixels.Apply();foreach(var pixel in pixels.GetPixels())if(!float.IsFinite(pixel.r)||!float.IsFinite(pixel.g)||!float.IsFinite(pixel.b))throw new Exception("Nonfinite fitted HDR capture");var c=pixels.GetPixel(8,8);var gpu=new Vector3(c.r,c.g,c.b);var expected=Vector3.Scale(albedo,input.samples[i].expectedClampedEOverPi);maxError=Mathf.Max(maxError,Mathf.Abs(gpu.x-expected.x),Mathf.Abs(gpu.y-expected.y),Mathf.Abs(gpu.z-expected.z));rows.Add(new Row{sample=i,albedoCase=a,normal=n,albedo=albedo,cpuSignedEOverPi=cpuSigned,gpuOutgoingRGB=gpu,expectedOutgoingRGB=expected});File.WriteAllBytes(output+"Fitted-"+i.ToString("D2")+"-Albedo"+a+".exr",pixels.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));camera.targetTexture=null;RenderTexture.active=previous;UnityEngine.Object.DestroyImmediate(pixels);UnityEngine.Object.DestroyImmediate(rt);
   }
  }
  File.WriteAllText(output+"proof.json",JsonUtility.ToJson(new Proof{status=float.IsFinite(maxError)&&maxError<.00001f?"PassedFittedNeutralShaderOnly":"RejectedFittedShaderResponse",unityVersion=Application.unityVersion,driver=SystemInfo.graphicsDeviceVersion,shader=shader.name,maxAbsError=maxError,rows=rows.ToArray()},true));if(!float.IsFinite(maxError)||maxError>=.00001f)throw new Exception("Fitted SH native response differs "+maxError);Debug.Log("CS_FITTED_SOURCE_SH_NEUTRAL_GPU_OK; actual coefficient packing and albedo response only, not actor or source L2 fidelity acceptance");EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
 }
 static bool Finite(Vector3 v)=>float.IsFinite(v.x)&&float.IsFinite(v.y)&&float.IsFinite(v.z);
}
