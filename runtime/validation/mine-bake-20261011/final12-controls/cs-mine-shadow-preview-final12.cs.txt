using System;using System.IO;using System.Linq;using System.Collections.Generic;using System.Security.Cryptography;using UnityEngine;using UnityEngine.Rendering;using UnityEditor;using UnityEditor.SceneManagement;using CriticalShift.Player.Editor;using CriticalShift.Features.Player.Unity;
public static class CsMineShadowPreviewFinal12 {
 [Serializable] public class ObjectRecord {public string name;public bool visibleShadow;}
 [Serializable] public class Inventory {public ObjectRecord[] objects;public string sourceSha256;}
 [Serializable] public class LightRecord {public string name,type,bakeMode,casterMode;public int shadowMaskChannel;public float intensity,range;}
 public static void Render(){
  string output="/tmp/cs-mine-shadow-preview-final12/";Directory.CreateDirectory(output);EditorSceneManager.OpenScene(SpawnMapBuild.ScenePath);var map=JsonUtility.FromJson<ExportMap>(File.ReadAllText(SpawnMapBuild.Folder+"map-bindings.json"));var package=map.materials.Single(p=>p.room=="mine");var root=GameObject.Find("ROOM_mine").transform;var inventory=JsonUtility.FromJson<Inventory>(File.ReadAllText("/tmp/cs-mine-source-inventory-final12/inventory.json"));var excluded=inventory.objects.Where(o=>!o.visibleShadow).Select(o=>"CS_mine_"+o.name).ToHashSet();var casters=root.GetComponentsInChildren<MeshRenderer>().Where(r=>excluded.Contains(r.name)).ToArray();if(casters.Length!=24)throw new Exception("Expected24 actual hanging lantern renderers: "+casters.Length);
  File.WriteAllText(output+"source-shadow-exclusions.json",JsonUtility.ToJson(inventory,true));File.WriteAllLines(output+"native-excluded-renderers.txt",casters.Select(r=>r.name+" before="+r.shadowCastingMode));File.WriteAllLines(output+"actual-native-lights.jsonl",UnityEngine.Object.FindObjectsByType<Light>().Select(l=>JsonUtility.ToJson(new LightRecord{name=l.name,type=l.type.ToString(),bakeMode=l.lightmapBakeType.ToString(),casterMode=l.lightShadowCasterMode.ToString(),shadowMaskChannel=l.bakingOutput.occlusionMaskChannel,intensity=l.intensity,range=l.range})));
  foreach(var camera in UnityEngine.Object.FindObjectsByType<Camera>())camera.enabled=false;GameObject.Find("Local crew worker").SetActive(false);Vector3 V(float[] a)=>new Vector3(a[0],a[2],a[1]);var views=JsonUtility.FromJson<SourceViews>(File.ReadAllText("/workspace/critical-shift-spawn-runtime/runtime/out/mine-bake-audit/cameras.json"));var view=views.views.Single(v=>v.name=="R39_02_TUNNEL");
  foreach(string mode in new[]{"before","source-shadow-flags"}){
   if(mode!="before")foreach(var r in casters)r.shadowCastingMode=ShadowCastingMode.Off;
   var node=new GameObject("Shadow preview");var camera=node.AddComponent<Camera>();camera.transform.SetPositionAndRotation(root.TransformPoint(V(view.position)),root.rotation*Quaternion.LookRotation(V(view.forward),V(view.up)));camera.fieldOfView=view.verticalFov;camera.nearClipPlane=.05f;camera.farClipPlane=250;camera.allowHDR=true;camera.allowMSAA=false;camera.clearFlags=CameraClearFlags.SolidColor;var sky=package.assemblyLighting;camera.backgroundColor=new Color(sky.world_color[0],sky.world_color[1],sky.world_color[2]).gamma*sky.world_strength;camera.renderingPath=RenderingPath.Forward;MineVisualLighting.ConfigureAtmosphere(camera,package);
   var rt=new RenderTexture(640,360,24,RenderTextureFormat.ARGBHalf,RenderTextureReadWrite.Linear);camera.targetTexture=rt;camera.Render();var previous=RenderTexture.active;RenderTexture.active=rt;var linear=new Texture2D(640,360,TextureFormat.RGBAHalf,false,true);linear.ReadPixels(new Rect(0,0,640,360),0,0);linear.Apply();foreach(var c in linear.GetPixels())if(!float.IsFinite(c.r)||!float.IsFinite(c.g)||!float.IsFinite(c.b))throw new Exception("Nonfinite preview");File.WriteAllBytes(output+mode+".exr",linear.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));camera.targetTexture=null;RenderTexture.active=previous;UnityEngine.Object.DestroyImmediate(linear);UnityEngine.Object.DestroyImmediate(rt);
   node.AddComponent<BlenderView>().Configure(SpawnSurfaceBuild.View(package));rt=new RenderTexture(640,360,24,RenderTextureFormat.ARGB32,RenderTextureReadWrite.sRGB);camera.targetTexture=rt;camera.Render();RenderTexture.active=rt;var pixels=new Texture2D(640,360,TextureFormat.RGB24,false,false);pixels.ReadPixels(new Rect(0,0,640,360),0,0);pixels.Apply();File.WriteAllBytes(output+mode+".png",pixels.EncodeToPNG());camera.targetTexture=null;RenderTexture.active=previous;UnityEngine.Object.DestroyImmediate(pixels);UnityEngine.Object.DestroyImmediate(rt);UnityEngine.Object.DestroyImmediate(node);
  }
  Debug.Log("CS_SOURCE_LANTERN_SHADOW_PREVIEW_OK; current baked baseline retained;24 source exclusions; no scene/source saved; no fresh bake acceptance");EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
 }
}
