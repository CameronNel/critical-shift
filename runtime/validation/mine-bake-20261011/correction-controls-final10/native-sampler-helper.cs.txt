public static class CsCliffNativeIsolation {
 public static void Render() {
  EditorSceneManager.OpenScene(SpawnMapBuild.ScenePath); LightProbes.Tetrahedralize();ReflectionProbe.UpdateCachedState();
  var root=GameObject.Find("ROOM_mine").transform;
  var actual=UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None).Single(r=>r.name=="CS_front-end-area_cliff_face");
  var original=actual.sharedMaterial;
  Debug.Log("CS_CLIFF_ACTUAL material="+original.name+" main="+AssetDatabase.GetAssetPath(original.mainTexture)+" normal="+AssetDatabase.GetAssetPath(original.GetTexture("_BumpMap"))+" lightmap="+actual.lightmapIndex+" offset="+actual.lightmapScaleOffset+" mesh="+AssetDatabase.GetAssetPath(actual.GetComponent<MeshFilter>().sharedMesh));

  foreach(var c in UnityEngine.Object.FindObjectsByType<Camera>(FindObjectsSortMode.None))c.enabled=false;
  var v=JsonUtility.FromJson<SourceViews>(File.ReadAllText("/workspace/critical-shift-spawn-runtime/runtime/out/mine-bake-audit/cameras.json")).views[0];
  Vector3 V(float[] a)=>new Vector3(a[0],a[2],a[1]);
  var node=new GameObject("Diagnostic cliff camera");node.transform.SetPositionAndRotation(root.TransformPoint(V(v.position)),root.rotation*Quaternion.LookRotation(V(v.forward),V(v.up)));
  var camera=node.AddComponent<Camera>();camera.allowHDR=true;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.black;camera.nearClipPlane=.05f;camera.farClipPlane=250;camera.fieldOfView=v.verticalFov;camera.renderingPath=RenderingPath.Forward;
  var folder="/workspace/critical-shift-spawn-runtime/runtime/out/mine-bake-audit/wood-production-sampler-final10/";Directory.CreateDirectory(folder);
  var albedo=new Material(original);albedo.SetColor("_Color",Color.black);albedo.SetFloat("_DielectricSpecular",0);albedo.SetFloat("_Metallic",0);albedo.DisableKeyword("_METALLICGLOSSMAP");albedo.DisableKeyword("_NORMALMAP");albedo.EnableKeyword("_EMISSION");albedo.SetTexture("_EmissionMap",original.mainTexture);albedo.SetColor("_EmissionColor",Color.white);
  var neutral=new Material(original);neutral.SetTexture("_MainTex",Texture2D.whiteTexture);neutral.SetColor("_Color",new Color(.18f,.18f,.18f).gamma);neutral.SetColor("_EmissionColor",Color.black);neutral.DisableKeyword("_EMISSION");
  void Capture(string name,Material material){actual.sharedMaterial=material;var target=new RenderTexture(640,360,24,RenderTextureFormat.ARGBHalf,RenderTextureReadWrite.Linear);camera.targetTexture=target;camera.Render();RenderTexture.active=target;var p=new Texture2D(640,360,TextureFormat.RGBAHalf,false,true);p.ReadPixels(new Rect(0,0,640,360),0,0);p.Apply();File.WriteAllBytes(folder+name+".exr",p.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));RenderTexture.active=null;camera.targetTexture=null;UnityEngine.Object.DestroyImmediate(p);UnityEngine.Object.DestroyImmediate(target);Debug.Log("CS_CLIFF_ISOLATION_CAPTURE "+name);}
  Capture("baseline",original);
  var package=JsonUtility.FromJson<ExportMap>(File.ReadAllText(SpawnMapBuild.Folder+"map-bindings.json")).materials.Single(p=>p.room=="mine");
  SpawnSurfaceBuild.Materials(package);
  foreach(var m in UnityEngine.Object.FindObjectsByType<MeshRenderer>(FindObjectsSortMode.None).SelectMany(r=>r.sharedMaterials).Where(m=>m.IsKeywordEnabled("_SOURCE_WOOD")).Distinct()){var t=(Texture2D)m.GetTexture("_WoodHigh");if(t.format!=TextureFormat.BC7 || t.anisoLevel!=0)throw new Exception("Coefficient sampler import changed: "+t.format+" aniso="+t.anisoLevel);Debug.Log("CS_WOOD_HIGH_PRODUCTION_BC7_ANISO0="+AssetDatabase.GetAssetPath(t)+" size="+t.width+" mips="+t.mipmapCount+" globalAniso="+QualitySettings.anisotropicFiltering);}
  Capture("production-import",original);
  actual.lightmapIndex=-1;actual.lightProbeUsage=UnityEngine.Rendering.LightProbeUsage.Off;actual.reflectionProbeUsage=UnityEngine.Rendering.ReflectionProbeUsage.Off;
  foreach(var view in JsonUtility.FromJson<SourceViews>(File.ReadAllText("/workspace/critical-shift-spawn-runtime/runtime/out/mine-bake-audit/cameras.json")).views){node.transform.SetPositionAndRotation(root.TransformPoint(V(view.position)),root.rotation*Quaternion.LookRotation(V(view.forward),V(view.up)));camera.fieldOfView=view.verticalFov;Capture("production-import-cliff-adapter-"+view.name,original);}
  actual.sharedMaterial=original;Debug.Log("CS_CLIFF_ISOLATION_OK; diagnostic only, scene not saved");
 }
 static void SunRadiometry() {
  var folder="Assets/CriticalShift/Generated/MineSunRadiometry";
  if(AssetDatabase.IsValidFolder(folder))throw new Exception("Radiometry scratch already exists");
  AssetDatabase.CreateFolder("Assets/CriticalShift/Generated","MineSunRadiometry");
  try {
   EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);
   RenderSettings.skybox=null;RenderSettings.ambientMode=UnityEngine.Rendering.AmbientMode.Flat;RenderSettings.ambientLight=Color.black;RenderSettings.ambientIntensity=0;RenderSettings.reflectionIntensity=0;
   var quad=GameObject.CreatePrimitive(PrimitiveType.Quad);var renderer=quad.GetComponent<MeshRenderer>();
   var mesh=UnityEngine.Object.Instantiate(quad.GetComponent<MeshFilter>().sharedMesh);mesh.uv2=mesh.uv;AssetDatabase.CreateAsset(mesh,folder+"/Quad.asset");quad.GetComponent<MeshFilter>().sharedMesh=mesh;
   GameObjectUtility.SetStaticEditorFlags(quad,StaticEditorFlags.ContributeGI);renderer.receiveGI=ReceiveGI.Lightmaps;
   var material=new Material(Shader.Find("CriticalShift/BlenderSurface"));material.SetColor("_Color",new Color(.18f,.18f,.18f).gamma);material.SetFloat("_DielectricSpecular",0);material.SetFloat("_Metallic",0);material.SetFloat("_Glossiness",.5f);renderer.sharedMaterial=material;
   var sun=new GameObject("Flat radiometry white sun").AddComponent<Light>();sun.type=LightType.Directional;sun.color=Color.white;sun.shadows=LightShadows.None;sun.renderMode=LightRenderMode.ForcePixel;sun.lightmapBakeType=LightmapBakeType.Realtime;sun.intensity=1.6f;
   var camera=new GameObject("Flat radiometry camera").AddComponent<Camera>();camera.transform.position=new Vector3(0,0,-3);camera.orthographic=true;camera.orthographicSize=.4f;camera.clearFlags=CameraClearFlags.SolidColor;camera.backgroundColor=Color.black;camera.allowHDR=true;camera.renderingPath=RenderingPath.Forward;
   void Save(string name){var rt=new RenderTexture(128,128,24,RenderTextureFormat.ARGBFloat,RenderTextureReadWrite.Linear);camera.targetTexture=rt;camera.Render();RenderTexture.active=rt;var p=new Texture2D(128,128,TextureFormat.RGBAFloat,false,true);p.ReadPixels(new Rect(0,0,128,128),0,0);p.Apply();File.WriteAllBytes("/workspace/critical-shift-spawn-runtime/runtime/out/mine-bake-audit/"+name+".exr",p.EncodeToEXR(Texture2D.EXRFlags.OutputAsFloat));RenderTexture.active=null;camera.targetTexture=null;UnityEngine.Object.DestroyImmediate(p);UnityEngine.Object.DestroyImmediate(rt);}
   Save("sun-radiometry-native-source-energy");sun.intensity=1.6f/Mathf.PI;Save("sun-radiometry-native-energy-over-pi");
   var settings=new LightingSettings {bakedGI=true,realtimeGI=false,lightmapper=LightingSettings.Lightmapper.ProgressiveCPU,lightmapResolution=32,lightmapMaxSize=128,lightmapPadding=4,directSampleCount=64,indirectSampleCount=8,environmentSampleCount=8,maxBounces=1,lightmapCompression=LightmapCompression.None};
   AssetDatabase.CreateAsset(settings,folder+"/Lighting.lighting");Lightmapping.lightingSettings=settings;sun.lightmapBakeType=LightmapBakeType.Baked;
   EditorSceneManager.SaveScene(EditorSceneManager.GetActiveScene(),folder+"/Scene.unity");
   if(!Lightmapping.Bake() || renderer.lightmapIndex<0 || renderer.lightmapIndex>=LightmapSettings.lightmaps.Length)throw new Exception("Radiometry native bake failed");
   Save("sun-radiometry-native-baked-energy-over-pi");Debug.Log("CS_SUN_RADIOMETRY_NATIVE_OK rendererLightmap="+renderer.lightmapIndex+" driver="+SystemInfo.graphicsDeviceName+" version="+SystemInfo.graphicsDeviceVersion);
  }finally{EditorSceneManager.NewScene(NewSceneSetup.EmptyScene,NewSceneMode.Single);AssetDatabase.DeleteAsset(folder);}
 }
}
