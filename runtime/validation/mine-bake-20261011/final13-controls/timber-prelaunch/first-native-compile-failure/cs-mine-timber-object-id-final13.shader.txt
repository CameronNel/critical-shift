Shader "CriticalShift/Diagnostics/TimberObjectIdFinal13"
{
 Properties {_MainTex("Texture",2D)="white"{} _Color("Tint",Color)=(1,1,1,1) _Cutoff("Cutoff",Float)=.5 _CsSelected("Exact object ID",Float)=0}
 CGINCLUDE
 #include "UnityCG.cginc"
 sampler2D _MainTex;float4 _MainTex_ST,_Color;float _Cutoff,_CsSelected;
 struct v2f {float4 position:SV_POSITION;float2 uv:TEXCOORD0;};
 v2f vert(appdata_base v){v2f o;o.position=UnityObjectToClipPos(v.vertex);o.uv=TRANSFORM_TEX(v.texcoord,_MainTex);return o;}
 float4 opaque(v2f i):SV_Target{return float4(_CsSelected,_CsSelected,_CsSelected,1);}
 float4 transparent(v2f i):SV_Target{return float4(_CsSelected,_CsSelected,_CsSelected,tex2D(_MainTex,i.uv).a*_Color.a);}
 float4 cutout(v2f i):SV_Target{clip(tex2D(_MainTex,i.uv).a*_Color.a-_Cutoff);return opaque(i);}
 ENDCG
 SubShader {Tags {"RenderType"="Opaque"} Pass {Cull Off ZWrite On ZTest LEqual CGPROGRAM
 #pragma vertex vert
 #pragma fragment opaque
 ENDCG}}
 SubShader {Tags {"RenderType"="TransparentCutout" "Queue"="AlphaTest"} Pass {Cull Off ZWrite On ZTest LEqual CGPROGRAM
 #pragma vertex vert
 #pragma fragment cutout
 ENDCG}}
 SubShader {Tags {"RenderType"="Transparent" "Queue"="Transparent"} Pass {Cull Off ZWrite Off ZTest LEqual Blend SrcAlpha OneMinusSrcAlpha CGPROGRAM
 #pragma vertex vert
 #pragma fragment transparent
 ENDCG}}
}
