def material(name,color,rough=.7,metal=0,broad=0,emission=0):
    m=bpy.data.materials.get(name) or bpy.data.materials.new(name)
    m.diffuse_color=(*rgb(color),1); m.use_nodes=True
    n=m.node_tree.nodes; n.clear(); p=n.new('ShaderNodeBsdfPrincipled'); out=n.new('ShaderNodeOutputMaterial')
    p.inputs['Base Color'].default_value=m.diffuse_color
    p.inputs['Roughness'].default_value=rough; p.inputs['Metallic'].default_value=metal
    p.inputs['Emission Color'].default_value=m.diffuse_color; p.inputs['Emission Strength'].default_value=emission
    m.node_tree.links.new(p.outputs[0],out.inputs['Surface'])
    if broad:
        tex=n.new('ShaderNodeTexNoise'); tex.inputs['Scale'].default_value=3.5
        tex.inputs['Detail'].default_value=1; tex.inputs['Roughness'].default_value=.4
        ramp=n.new('ShaderNodeValToRGB')
        ramp.color_ramp.elements[0].position=.15; ramp.color_ramp.elements[1].position=.85
        for e,k in zip(ramp.color_ramp.elements,(1-broad,1+broad)):
            e.color=(*[min(1,c*k) for c in rgb(color)],1)
        m.node_tree.links.new(tex.outputs['Fac'],ramp.inputs[0]); m.node_tree.links.new(ramp.outputs[0],p.inputs['Base Color'])
    return m

def plate(name,pos,w,h,depth,mat,parent=None,cut=.03):
    # Folded manufactured plate: chamfered XZ profile, extruded along Y.
    c=min(cut,w/5,h/5)
    p=[(-w/2+c,-h/2),(w/2-c,-h/2),(w/2,-h/2+c),(w/2,h/2-c),(w/2-c,h/2),(-w/2+c,h/2),(-w/2,h/2-c),(-w/2,-h/2+c)]
    v=[(x,y,z) for y in (-depth/2,depth/2) for x,z in p]
    f=[tuple(range(8)),tuple(range(15,7,-1))]+[(i+8,(i+1)%8+8,(i+1)%8,i) for i in range(8)]
    return mesh(name,v,f,mat,parent,pos)
def cylinder(name,pos,r,depth,mat,parent=None,axis='Z',vertices=16):
    v=[]
    for z in (-depth/2,depth/2):
        for i in range(vertices):
            a=2*math.pi*i/vertices;p=(r*math.cos(a),r*math.sin(a),z)
            if axis=='Y':p=(p[0],p[2],p[1])
            if axis=='X':p=(p[2],p[0],p[1])
            v.append(p)
    faces=[tuple(range(vertices-1,-1,-1)),tuple(range(vertices,2*vertices))]+[(i,(i+1)%vertices,(i+1)%vertices+vertices,i+vertices) for i in range(vertices)]
    if axis=='Y':faces=[tuple(reversed(f)) for f in faces]
    return mesh(name,v,faces,mat,parent,pos)
def profile(name,outline,depth,mat,parent,axis='X'):
    """Purpose-built cast/pressed cross-section, extruded on the declared axis."""
    v=[];n=len(outline)
    for end in (-depth/2,depth/2):
        for a,b in outline:v.append((end,a,b) if axis=='X' else (a,end,b))
    f=[tuple(range(n-1,-1,-1)),tuple(range(n,n*2))]+[(i,(i+1)%n,(i+1)%n+n,i+n) for i in range(n)]
    return mesh(name,v,f,mat,parent)

def upholstery(o,mat,fold=False):
    """Authored soft surface and wrapped sides, inside the inherited envelope."""
    co=[v.co for v in o.data.vertices];lo=[min(v[i] for v in co) for i in range(3)];hi=[max(v[i] for v in co) for i in range(3)]
    n=12;verts=[];faces=[]
    def vertex(x,y,z):return tuple(lo[i]+(hi[i]-lo[i])*(q+1)/2 for i,q in enumerate((x,y,z)))
    for bottom in (False,True):
        for i in range(n+1):
            x=-1+2*i/n
            for j in range(n+1):
                y=-1+2*j/n
                xx=x*(1-.07*abs(y)**4);yy=y*(1-.07*abs(x)**4)
                shape=max(0,(1-x*x)*(1-y*y))**.32
                z=-.1-.9*shape if bottom else .1+.9*shape
                if fold and not bottom:z=max(.1,z-.28*math.sin(2*math.pi*x+y)**2*(1-x*x)*(1-y*y))
                verts.append(vertex(xx,yy,z))
    size=(n+1)**2
    for b in range(2):
        for i in range(n):
            for j in range(n):
                q=b*size+i*(n+1)+j;face=(q,q+1,q+n+2,q+n+1)
                faces.append(tuple(reversed(face)) if b==0 else face)
    edge=list(range(n+1))+[i*(n+1)+n for i in range(1,n+1)]+[n*(n+1)+j for j in range(n-1,-1,-1)]+[i*(n+1) for i in range(n-1,0,-1)]
    for a,b in zip(edge,edge[1:]+edge[:1]):faces.append((a,b,b+size,a+size))
    md=bpy.data.meshes.new('MED_R2 | Upholstery '+o.name);md.from_pydata(verts,[],faces);md.update();md.materials.append(mat)
    for p in md.polygons:p.use_smooth=True
    o.data=md
    for mod in list(o.modifiers):
        if mod.type in {'BEVEL','WEIGHTED_NORMAL'}:o.modifiers.remove(mod)
