def controlled_finish(mat,base_color,wear_color,roughness):
    """Purpose-painted wear, not a texture-noise replacement for construction."""
    nt=mat.node_tree;nodes=nt.nodes;links=nt.links;p=principal(mat)
    attr=nodes.new('ShaderNodeAttribute');attr.attribute_name='med_contact_wear'
    mix=nodes.new('ShaderNodeMixRGB');mix.inputs[1].default_value=(*rgb(base_color),1);mix.inputs[2].default_value=(*rgb(wear_color),1)
    links.new(attr.outputs['Fac'],mix.inputs[0]);links.new(mix.outputs[0],p.inputs['Base Color'])
    rough=nodes.new('ShaderNodeMapRange');rough.inputs['To Min'].default_value=roughness;rough.inputs['To Max'].default_value=min(.98,roughness+.12)
    links.new(attr.outputs['Fac'],rough.inputs[0]);links.new(rough.outputs[0],p.inputs['Roughness'])
    coord=nodes.new('ShaderNodeUVMap');coord.uv_map='MED_Physical_1m';noise=nodes.new('ShaderNodeTexNoise')

    links.new(coord.outputs[0],noise.inputs['Vector'])
    bump=nodes.new('ShaderNodeBump');bump.inputs['Strength'].default_value=.055;bump.inputs['Distance'].default_value=.00015
    links.new(noise.outputs['Fac'],bump.inputs['Height']);links.new(bump.outputs[0],p.inputs['Normal'])

def unitq(co,lo,sp):
    return Vector(tuple((co[i]-lo[i])/sp[i] for i in range(3)))

def bounds(o):
    lo=Vector(tuple(min(v.co[i] for v in o.data.vertices) for i in range(3)))
    hi=Vector(tuple(max(v.co[i] for v in o.data.vertices) for i in range(3)))
    return lo,hi

def folded_skin(o,axis,sign,inset=.08,depth=.30):
    """A folded sheet with an eight-sided recessed skin and exposed return.

    Parameters are fractions of the measured mesh envelope. The outer extents
    remain exact, including opposing rear faces; this is not a resized panel.
    """
    lo,hi=bounds(o);sp=hi-lo;other=[i for i in range(3) if i!=axis];v=[];f=[];mi=[]
    outline=[(-1,-.90),(-.90,-1),(.90,-1),(1,-.90),(1,.90),(.90,1),(-.90,1),(-1,.90)]
    for scale,z in [(1,sign),(1-inset,sign),(.98-inset,sign*(1-depth)),(1,-sign)]:
        for a,b in outline:
            p=(lo+hi)/2;p[other[0]]+=a*sp[other[0]]*.5*scale;p[other[1]]+=b*sp[other[1]]*.5*scale;p[axis]+=z*sp[axis]*.5;v.append(tuple(p))
    for ring,mat in [(0,0),(1,1)]:
        for i in range(8):f.append((ring*8+i,ring*8+(i+1)%8,(ring+1)*8+(i+1)%8,(ring+1)*8+i));mi.append(mat)
    f.append(tuple(range(16,24)));mi.append(0)
    f.append(tuple(range(31,23,-1)));mi.append(2)
    for i in range(8):f.append((i,(i+1)%8,24+(i+1)%8,24+i));mi.append(2)
    replace(o,v,f,[M['ivory'],M['navy'],M['edge']],mi)
    # The bound-facing plane is the quiet sheet skin; ageing gathers at lower
    # seams and handled edges, rather than across every square centimetre.
    attr=o.data.attributes.new('med_contact_wear','FLOAT','POINT')
    for vertex,a in zip(o.data.vertices,attr.data):
        q=unitq(vertex.co,lo,sp);a.value=min(.8,max(0,(.25-q.z)*2)+(.20 if abs(q[other[0]]-.5)>.40 else 0))

def pressed_liner(o,centre=False):
    lo,hi=bounds(o);sp=hi-lo;v=[];f=[]
    # Rib channels stiffen a pressed steel liner. No repeated picture-frame rim.
    xs=[0,.10,.13,.16,.84,.87,.90,1] if not centre else [0,.47,.49,.51,.53,1]
    for rear in (False,True):
        for x in xs:
            for z in (0,1):
                depth=1 if rear else (.28 if (.13<=x<=.16 or .84<=x<=.87) and not centre else .08)
                v.append((lo.x+sp.x*x,lo.y+sp.y*depth,lo.z+sp.z*z))
    size=len(xs)*2
    for rear in range(2):
        for i in range(len(xs)-1):
            q=rear*size+i*2;f.append((q,q+2,q+3,q+1))
    for i in range(len(xs)-1):
        q=i*2;f.extend([(q,q+size,q+size+2,q+2),(q+1,q+3,q+size+3,q+size+1)])
    f.extend([(0,1,size+1,size),(size-2,2*size-2,2*size-1,size-1)])
    # Exact front datum at the folded outer return.
    v[0]=(v[0][0],lo.y,v[0][2]);v[1]=(v[1][0],lo.y,v[1][2])
    replace(o,v,f,[M['ivory']])

def piston_stage(o):
    lo,hi=bounds(o);sp=hi-lo;v=[];f=[];n=48
    for z,r in [(lo.z,1),(lo.z+sp.z*.10,.90),(hi.z-sp.z*.10,.90),(hi.z,1)]:
        for i in range(n):
            a=i*math.tau/n;v.append(((lo.x+hi.x)/2+sp.x*.5*r*math.cos(a),(lo.y+hi.y)/2+sp.y*.5*r*math.sin(a),z))
    f=[tuple(range(n-1,-1,-1)),tuple(range(3*n,4*n))]
    for k in range(3):
        for i in range(n):f.append((k*n+i,k*n+(i+1)%n,(k+1)*n+(i+1)%n,(k+1)*n+i))
    replace(o,v,f,[M['steel']],smooth=True)
    for face in o.data.polygons:face.use_smooth=len(face.vertices)==4

def tailored_pad(o,pillow=False):
    """Bound cover with flat working area, corner darts and shallow tension folds."""
    lo,hi=bounds(o);sp=hi-lo;v=[];f=[];n=36;m=36
    for bottom in (False,True):
        for i in range(n+1):
            x=-1+2*i/n
            for j in range(m+1):
                y=-1+2*j/m
                # Crisp sewn corners instead of a globally inflated blob.
                xx=x*(1-.025*abs(y)**12);yy=y*(1-.025*abs(x)**12)
                edge=max(abs(x),abs(y));rise=max(0,1-edge**10)**.5
                z=-1 if bottom else -.12+1.12*rise
                if pillow and not bottom:
                    z-=.33*(math.exp(-((x-.20)/.45)**2-((y+.12)/.60)**2))
                    z-=.15*math.sin(8*x+3*y)**2*(abs(x)**3+abs(y)**3)*rise
                elif not bottom:
                    z-=.075*math.sin(18*y+4*x)**2*abs(x)**5*rise
                    z-=.10*math.exp(-((x+.28)/.20)**2)*math.exp(-((y+.8)/.17)**2)
                v.append(tuple((lo+hi)/2+Vector((xx*sp.x/2,yy*sp.y/2,z*sp.z/2))))
    size=(n+1)*(m+1)
    for b in range(2):
        for i in range(n):
            for j in range(m):
                q=b*size+i*(m+1)+j;face=(q,q+1,q+m+2,q+m+1);f.append(tuple(reversed(face)) if b==0 else face)
    edge=list(range(m+1))+[i*(m+1)+m for i in range(1,n+1)]+[n*(m+1)+j for j in range(m-1,-1,-1)]+[i*(m+1) for i in range(n-1,0,-1)]
    for a,b in zip(edge,edge[1:]+edge[:1]):f.append((a,b,b+size,a+size))
    # Recover exact extrema without changing the object origin or transform.
    for k in range(3):
        amin=min(p[k] for p in v);amax=max(p[k] for p in v)
        v=[tuple(lo[t]+(p[t]-amin)/(amax-amin)*sp[t] if t==k else p[t] for t in range(3)) for p in v]
    replace(o,v,f,[M['linen'] if pillow else M['pad']],smooth=True)
    obsolete={'Contoured head pad':['MED_R2 | OCRU upholstered seam construction'],'Recovery pillow':['MED_R2 | Recovery pillow wrapped seam']}.get(o.name,[])
    if o.name.startswith('Segmented adult berth cushion'):obsolete.append('MED_R2 | '+o.name+' bound seam')
    for old_name in obsolete:
        old=s.objects.get(old_name)
        if old:bpy.data.objects.remove(old,do_unlink=True)
        supports[:]=[r for r in supports if r['object']!=old_name]
    attr=o.data.attributes.new('med_contact_wear','FLOAT','POINT')
    for vert,a in zip(o.data.vertices,attr.data):
        q=unitq(vert.co,lo,sp)
        a.value=.10+.22*math.exp(-((q.x-.32)/.18)**2-((q.y-.18)/.14)**2)+.20*max(0,(abs(q.x-.5)-.35)/.15)
    def seams():
        bpy.context.view_layer.update()
        tree=BVHTree.FromPolygons([o.matrix_world@vert.co for vert in o.data.vertices],[tuple(poly.vertices) for poly in o.data.polygons])
        edge=list(range(m+1))+[i*(m+1)+m for i in range(1,n+1)]+[n*(m+1)+j for j in range(m-1,-1,-1)]+[i*(m+1) for i in range(n-1,0,-1)]
        pts=[]
        for ix in edge:
            hit,normal,index,distance=tree.find_nearest(o.matrix_world@Vector(v[ix]))
            pts.append(tuple(o.parent.matrix_world.inverted()@(hit+normal*.0004)))
        line('Bound upholstery welt',pts+[pts[0]],.0012,M['suture'] if not pillow else M['seam'],o.parent)
        # Stitch run follows the actual sewn cover, not a straight floating line.
        for ix in range(2,n-2,2):
            for jy in (1,m-1):
                p=Vector(v[ix*(m+1)+jy]);p.z+=.0005
                p2=Vector(v[(ix+1)*(m+1)+jy]);p2.z+=.0005
                line('Upholstery stitch', [tuple(o.matrix_local@p),tuple(o.matrix_local@p2)],.00045,M['suture'] if not pillow else M['seam'],o.parent)
    seam=group('Tailored cover construction '+o.name,seams,o.name)
    seam['contact_check_surface_all_vertices']=True

def lathe(name,pos,rings,mat,parent,segments=48):
    v=[];f=[]
    for z,r in rings:
        for i in range(segments):
            a=i*math.tau/segments;v.append((r*math.cos(a),r*math.sin(a),z))
    f.append(tuple(range(segments-1,-1,-1)))
    for k in range(len(rings)-1):
        for i in range(segments):f.append((k*segments+i,k*segments+(i+1)%segments,(k+1)*segments+(i+1)%segments,(k+1)*segments+i))
    f.append(tuple(range((len(rings)-1)*segments,len(rings)*segments)))
    o=mesh(name,v,f,mat,parent,pos)
    for p in o.data.polygons:
        if len(p.vertices)==4:p.use_smooth=True
    return o

def textile_relaxation():
    cloth=s.objects['Folded cleaning cloth'];lo,hi=bounds(cloth);sp=hi-lo
    # Use the approved spawn's actual folded-cotton construction as the shape
    # reference, owned and fitted to this existing cleaning-cloth envelope.
    with bpy.data.libraries.load(str(spawn_path),link=False) as (a,b):
        b.objects=['V_CART_towel_0_folded_fabric','V_CART_towel_0_fold_edge','V_CART_towel_0_fold_edge.001']
    ref=b.objects[0];rlo,rhi=bounds(ref);rsp=rhi-rlo
    edge_sources=b.objects[1:]
    vv=[tuple(lo+Vector(tuple((v.co[k]-rlo[k])/rsp[k]*sp[k] for k in range(3)))) for v in ref.data.vertices]
    ff=[tuple(p.vertices) for p in ref.data.polygons]
    replace(cloth,vv,ff,[M['linen']],smooth=True)
    cloth['geometry_reference']='Approved spawn V_CART_towel_0_folded_fabric, fitted to original envelope; owned mesh copy.'
    bpy.data.objects.remove(ref,do_unlink=True)
    old=s.objects.get('MED_R2 | Folded cleaning cloth seam construction')
    if old:bpy.data.objects.remove(old,do_unlink=True)
    supports[:]=[r for r in supports if r['object']!='MED_R2 | Folded cleaning cloth seam construction']
    # Existing three folds get asymmetric compressed creases and softer hems.
    for v in cloth.data.vertices:
        q=unitq(v.co,lo,sp);x=(q.x-.5)*2;y=(q.y-.5)*2
        if .015<q.z<.98:
            v.co.z-=.0035*math.sin(5*x+2*y)**2*(1-x*x)*(1-y*y)
    a=cloth.data.attributes.new('med_contact_wear','FLOAT','POINT')
    for v,w in zip(cloth.data.vertices,a.data):
        q=unitq(v.co,lo,sp);w.value=.06+.26*math.exp(-((q.x-.19)/.22)**2-((q.y-.71)/.30)**2)
    cloth.data.update()
    def approved_edges():
        tree=BVHTree.FromPolygons([v.co for v in cloth.data.vertices],[tuple(p.vertices) for p in cloth.data.polygons])
        for edge in edge_sources:
            points=[]
            for cp in edge.data.splines[0].points:
                p=lo+Vector(tuple((cp.co[k]-rlo[k])/rsp[k]*sp[k] for k in range(3)))
                surface,normal,index,distance=tree.find_nearest(p)
                points.append(tuple(surface+normal*.0006))
            line('Approved cotton rolled fold edge',points,.0011,M['seam'],cloth)
    group('Constructed cleaning cotton folded edges',approved_edges,cloth.name)
    for obj in edge_sources:bpy.data.objects.remove(obj,do_unlink=True)
    # Sewn folded hems, paper-thin bound edges and a cotton care tag.
    def hems():
        for layer in range(3):
            points=[]
            for j in range(13):
                p=cloth.data.vertices[layer*338+j].co.copy();p.x-=.0012
                points.append(tuple(p))
            line('Compressed folded cotton hem',points,.0007,M['seam'],cloth)
    # The baseline already has a physical bound edge on all three folds. Keep
    # that construction; duplicating it here produced overlapping seam tubing.
textile_relaxation()
